int alarmMode = 0; // 0=alle, 1=Mo-Fr, 2=Sa-So

enum State {
  HOME,
  SET_HOUR,
  SET_MIN,
  SET_MODE,
  ALARM_RING
};

State state = HOME;
State lastState = HOME;

// Timing
unsigned long lastDraw = 0;
unsigned long lastPressTime = 0;
unsigned long holdStart = 0;

// Alarm Trigger
bool alarmTriggered = false;

  Serial.begin(115200);

  Wire.begin(21, 22);
  rtc.begin();

;

  pinMode(BTN_MENU, INPUT_PULLUP);

  pinMode(BUZZER, OUTPUT);

  tft.begin();
  tft.setRotation(1);
  tft.fillScreen(ILI9341_BLACK);

void loop() {
  DateTime now = rtc.now();

  handleButtons();


  if (state != lastState || millis() - lastDraw &gt; 1000) {
    drawScreen(now);
    lastState = state;
    lastDraw = millis();
  }

  checkAlarm(now);

  // Alarm Sound
  if (state == ALARM_RING) {
    tone(BUZZER, 1000);
  } else {
    noTone(BUZZER);
  }

void drawScreen(DateTime now) {
  tft.fillScreen(ILI9341_BLACK);

  switch (state) {

    case HOME:
      drawHome(now);
      break;

    case SET_HOUR:
      drawSet("Stunde", alarmHour);
      break;

    case SET_MIN:
      drawSet("Minute", alarmMinute);
      break;

    case SET_MODE:
      drawMode();
      break;

    case ALARM_RING:
      tft.setTextSize(3);
      tft.setCursor(40, 100);
      tft.setTextColor(ILI9341_RED);
      tft.println("ALARM!");
      break;

bool pressed(int pin) {
  if (digitalRead(pin) == LOW) {
    if (millis() - lastPressTime &gt; 200) {
      lastPressTime = millis();
      holdStart = millis();
      return true;
    }
  }
  return false;
}


bool held(int pin) {
  if (digitalRead(pin) == LOW) {
    if (millis() - holdStart &gt; 800) {
      if (millis() - lastPressTime &gt; 80) {
        lastPressTime = millis();
        return true;
      }
    }
  }
  return false;
}

void handleButtons() {

  // MENU / CONFIRM
  if (pressed(BTN_MENU)) {
    if (state == HOME) state = SET_HOUR;
    else if (state == SET_HOUR) state = SET_MIN;
    else if (state == SET_MIN) state = SET_MODE;
    else if (state == SET_MODE) state = HOME;
    else if (state == ALARM_RING) state = HOME;
  }

//set hours
  if (state == SET_HOUR) {
    if (pressed(BTN_UP) || held(BTN_UP))
      alarmHour = (alarmHour + 1) % 24;

    if (pressed(BTN_DOWN) || held(BTN_DOWN))
      alarmHour = (alarmHour - 1 + 24) % 24;
  }

//Set minutes
  if (state == SET_MIN) {
    if (pressed(BTN_UP) || held(BTN_UP))
      alarmMinute = (alarmMinute + 1) % 60;

    if (pressed(BTN_DOWN) || held(BTN_DOWN))
      alarmMinute = (alarmMinute - 1 + 60) % 60;
  }

  // Modus
  if (state == SET_MODE) {
    if (pressed(BTN_UP) || pressed(BTN_DOWN))
      alarmMode = (alarmMode + 1) % 3;
  }
}

void checkAlarm(DateTime now) {
  int d = now.dayOfTheWeek();

  bool weekday = (d &gt;= 1 &amp;&amp; d &lt;= 5);
  bool weekend = (d == 0 || d == 6);

  bool modeOk =
    (alarmMode == 0) ||
    (alarmMode == 1 &amp;&amp; weekday) ||
    (alarmMode == 2 &amp;&amp; weekend);

  if (now.hour() == alarmHour &amp;&amp;
      now.minute() == alarmMinute &amp;&amp;
      modeOk &amp;&amp;
      !alarmTriggered) {

    state = ALARM_RING;
    alarmTriggered = true;
  }

  if (now.minute() != alarmMinute) {
    alarmTriggered = false;
  }
}

# 4/13/2026 8 PM - 13.04.2026 Choosing of Components and techinical Drawing of Case

_Time spent: 2.5h_

First, I created a list of all the components I needed and checked their measurements, including thickness, height, width, and other dimensions. Based on these measurements, I created a rough sketch of the design I wanted.

I also decided to split the case into two parts a front cover and a back cover to make it easier to build

![Scan_Pic0106](https://stasis.hackclub-assets.com/images/1776110016662-i7vto4.jpg)

![Scan_Pic0105](https://stasis.hackclub-assets.com/images/1776108686546-or2m9m.jpg)


I fully sketched the relevent parts of the front cover. But i wasnt able to do them same for the back cover as i just didnt know how to do it, so I wrote down the  measurements and created simple front and side view sketches for it and left out the view from above .

After that i thought of ways to assemble and watched diferent videos. I then found some tiktoks in which somebody uses cylinders and puts the screw into the back where the screw head is stopped by material and the rest of the screw drills into the back cylinder.

I placed the cylinder at a location where i can just press the screnn between tehm to hold the screen so a press fit.

One of my first sketches was designed for the ILI9341 display module. But during the design process, I realized that the ST7789 display module is a better fit for my project, and even though both are the same kind of screen they have different dimensions so i needed to redesign the sketch because its 1:1 scale sketch.
.
![Screenshot 2026-04-13 190243](https://stasis.hackclub-assets.com/images/1776099776857-2r6ee4.png)

**As for the Parts these are the parts I need**

ESP32 Development Board (ESP32-WROOM-32)

2.8” TFT SPI Display 240x320 (ST7789 or ILI9341)

DS3231 RTC Module

Active Buzzer (3.3V–5V)

3x Push Buttons (Menu, Up, Down)

18650 Li-ion rechargeable battery (3.7V)

TP4056 Lithium Battery Charging Module (with protection)

MT3608 DC-DC Boost Converter (3.7V → 5V)

Jumper wires (Dupont cables)
